# Python translation of the Prolog knowledge base

# person(Name, date(Day, Month, Year))
people = {
    "arun": (12, 3, 2004),
    "bala": (25, 7, 2003),
    "chitra": (5, 11, 2004),
    "divya": (18, 3, 2005),
    "elango": (30, 1, 2003),
}

# dob(Name, D/M/Y)
def dob(name):
    if name in people:
        d, m, y = people[name]
        return f"{d}/{m}/{y}"
    return None

# born_in_year(Name, Year)
def born_in_year(year):
    return [name for name, date in people.items() if date[2] == year]

# born_in_month(Name, Month)
def born_in_month(month):
    return [name for name, date in people.items() if date[1] == month]

# older(P1, P2)
def older(p1, p2):
    if p1 not in people or p2 not in people or p1 == p2:
        return False
    # In python, comparing tuples directly (Year, Month, Day) evaluates chronologically.
    # An older person has a smaller/earlier birthdate tuple.
    d1, m1, y1 = people[p1]
    d2, m2, y2 = people[p2]
    return (y1, m1, d1) < (y2, m2, d2)

# Test queries
print("Date of Birth of Chitra:", dob("chitra"))
print("Born in 2004:", born_in_year(2004))
print("Born in March (Month 3):", born_in_month(3))
print("Is bala older than arun?", older("bala", "arun"))
